% Framing: We don't know how training shapes practice style
\begin{frame}[plain]{}
    \begin{quote}
        ...to what extent are physician beliefs about what constitutes clinically appropriate care primarily forged during training and to what extent are they potentially malleable to subsequent information...?
    \end{quote}
    \hfill{\small -- Finkelstein and Skinner, ``Regional Variations and Productivity Puzzles in Health Care"}
\end{frame}

\begin{frame}[plain]{}
    \textbf{What we know}
        \begin{itemize}
            \item A third of geographic variation in health care utilization is explained by physician treatment style \textcolor{gray}{\scriptsize\textcite{badinski2023}}
            \item Physicians adopt the practice style norms of their market after moving \textcolor{gray}{\scriptsize\textcite{molitor2018}}
        \end{itemize}

    \vspace{10mm}
    \textbf{What we don't know} 
        \begin{itemize}
            \item How much of practice style is set during training? 
        \end{itemize}
\end{frame}

\begin{frame}{This paper}
    We ask whether training environment has a lasting impact on later-career treatment decisions.

    \vspace{5mm}
    \begin{itemize}
        \item \textbf{Training environment:} Medical school HRR cath lab exposure
        \item \textbf{Treatment decision:} Invasive treatment in partial blockage heart attacks
        \item \textbf{Methodology:} Mover design
    \end{itemize}
\end{frame}

\begin{frame}{Literature review}
    Well-documented differences in physician practice style and health care utilization
        \begin{itemize}
            \item Evidenced by geographic variation in health care utilization \textcolor{gray}{\scriptsize\textcite{finkelstein2016}, \textcite{cutler2019}. \textcite{badinski2023}}
            \item Practice style evolution caused by peer effects \textcolor{gray}{\scriptsize\textcite{molitor2018}}, financial incentives \textcolor{gray}{\scriptsize\textcite{clemens_physician_2024}}, and new research \textcolor{gray}{\scriptsize\textcite{decicca_how_2024}}
        \end{itemize}

    \vspace{5mm}
    Limited evidence on the role of training in forming practice style
        \begin{itemize}
            \item Relies on rank as a proxy for quality \textcolor{gray}{\scriptsize\textcite{doyle_jr_returns_2010}, \textcite{schnell2018}}
            \item Variation not explained by residency program \textcolor{gray}{\scriptsize\textcite{epstein2009}}
        \end{itemize}
\end{frame}

% TODO: Add broader human capital implications
\begin{frame}{Contribution}
    Identify a specific channel through which medical school impacts practice
        \begin{itemize}
            \item Quantify medical school environment using exposure to procedural technology 
            \item Show that the training effect from that exposure lasts when controlling for rank
        \end{itemize}

    \vspace{5mm}
    Human capital formation: Initial training vs. accrued experience 
        \begin{itemize}
            \item Isolate training effect from current market effects
        \end{itemize}
\end{frame}

\begin{frame}{Preview of results}
    Medical school has a lasting impact
        \begin{itemize}
            \item 10pp increase in cath lab availability during training increases later cath rates by 0.58pp
            \item Effect is concentrated among interventional cardiologists
            \item Does not go away after controlling for other aspects (med school quality, residency)
        \end{itemize}

    \vspace{5mm}
    Peer effects compound the impact 
        \begin{itemize}
            \item If exposure rises for a whole peer group...
            \item 10pp increase in cath lab availability during training increases later cath rates by 0.97pp
        \end{itemize}
\end{frame}

\begin{frame}{Empirical strategy}
    \begin{equation*}
        y_{it} = \beta_\text{train}\tau_i + \theta\bar{y}_{-i,t} + \rho_{m(i)} + \delta_{d(i,t)} + \gamma_t + \epsilon_{it}
    \end{equation*}
    \begin{itemize}
        \item $y_{it}$ = mean residualized cath rate of cardiologist $i$ in year $t$
        \item $\tau_i$ = training-period cath lab share in cardiologist $i$'s medical school HRR
        \item $\bar{y}_{-i,t}$ = leave-one-out mean residualized cath rate of other cardiologists in cardiologist $i$'s practice HRR in year $t$
        \item $\rho_{m(i)}$ = medical school HRR fixed effect
        \item $\delta_{d(i,t)}$ = current practice HRR fixed effect
        \item $\gamma_t$ = year fixed effect
    \end{itemize}
\end{frame}

\begin{frame}{Empirical strategy}
    \textbf{Identifying variation:} Different cath lab exposure among cardiologists who attended medical school in the same HRR but graduated in different years

    \vspace{5mm}
    \textbf{Identifying assumption:} Within-HRR cath lab adoption is uncorrelated with changes in other cohort-level determinants of practice intensity
\end{frame}

\begin{frame}{Long-run effect with peer spillovers}
    Practice styles evolve due to peer effects \textcolor{gray}{\scriptsize\textcite{molitor2018}}
    
    \vspace{2.5mm}
    \hspace{30mm}{$\downarrow$}
    \vspace{2.5mm}

    What is the impact of training when taking peer effects into account?
    \begin{enumerate}
        \item Estimate peer effect using cardiologists who move mid-career
        \item Calculate the combined effect 
    \end{enumerate}
\end{frame}

\begin{frame}[label=peer]{Long-run effect with peer spillovers}
    \begin{equation*}
        \text{treatment style} = \beta_{train}\times(\text{training environment}) + \beta_{dest}(\text{peer's treatment styles}) + \text{error}
    \end{equation*}
    \vspace{2.5mm}
        \begin{equation*}
            \text{Long run effect:}\quad \Delta y^\ast = \frac{\beta_{train}\cdot\Delta\tau}{1-\beta_{dest}}
        \end{equation*}
    \begin{itemize}
        \item \textbf{Direct effect of training:} $\beta_{train} = 0.058$ 
        \item \textbf{Peer effect:} $\beta_{dest} = 0.40$ \quad\hyperlink{es}{\beamerbutton{Event study}} \hyperlink{es-direction}{\beamerbutton{By direction}}
        \item[$\rightarrow$] 10pp increase in training cath lab availability $\rightarrow$ 0.97pp increase in cath rate
    \end{itemize}
    
\end{frame}

\begin{frame}{Conclusion}
    Training effects persist throughout a physician's career, so later-career policies alone may not change practice patterns
        \begin{itemize}
            \item 1 SD increase in technology exposure shifts practice style one-third relative to the mean
            \item Training effect is amplified by peer-to-peer learning
        \end{itemize}

    \vspace{10mm}
    \textbf{Policy implication}
    \begin{itemize}
        \item Uniform technology access during training reduces practice style variation
    \end{itemize}
\end{frame}
